cmake_minimum_required(VERSION 3.20)
project(codons VERSION 0.1.0 LANGUAGES C)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Codons 0.1 requires 64-bit Windows (x86-64)")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(bio STATIC src/core.c src/parser.c src/analyze.c src/jit_win64.c src/runtime.c)
target_include_directories(bio PUBLIC include PRIVATE src)
target_compile_definitions(bio PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0601)
if(MSVC)
  target_compile_definitions(bio PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
add_executable(codons src/main.c)
target_link_libraries(codons PRIVATE bio)
enable_testing()
add_executable(bio_tests tests/test_bio.c)
target_include_directories(bio_tests PRIVATE src)
target_link_libraries(bio_tests PRIVATE bio)
foreach(target bio codons bio_tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
add_test(NAME bio_tests COMMAND bio_tests)
set_tests_properties(bio_tests PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}" TIMEOUT 60)
